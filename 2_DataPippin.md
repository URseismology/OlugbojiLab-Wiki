from flask import Flask, request

app = Flask(__name__)

@app.route('/data', methods=['POST'])
def post_data():
    raw_data = request.json
    # Process the data and return a response
    model = process_data(raw_data)
    return {'status': 'success', 'model': str(model)}

if __name__ == '__main__':
    app.run(debug=True)

# web_service.py
from spyne import Application, rpc, ServiceBase, Integer, Unicode
from spyne.protocol.soap import Soap11
from spyne.server.wsgi import WSGIServer

class DataService(ServiceBase):
    @rpc(Unicode, _returns=Unicode)
    def process_data(self, raw_data):
        # Process the data and return a response
        model = process_data(raw_data)
        return str(model)

application = Application([DataService], 'spyne.examples.hello.soap',
                          in_protocol=Soap11(validator='lxml'),
                          out_protocol=Soap11())

if __name__ == '__main__':
    from wsgiref.simple_server import make_server
    server = make_server('localhost', 8000, WSGIServer(application))
    server.serve_forever()
