```python
# Download_XML.ipynb

import obspy
from obspy import read_inventory
import os

# Function to download XML files
def download_xml(network, station):
    client = obspy.clients.fdsn.Client("IRIS")
    try:
        inv = client.get_stations(network=network, station=station, level="response")
        
        if not os.path.exists("data/xml_files"):
            os.makedirs("data/xml_files")

        xml_filename = f"data/xml_files/{network}_{station}.xml"
        inv.write(xml_filename, format="STATIONXML")
        print(f"Downloaded {xml_filename}")
    except Exception as e:
        print(f"Error downloading XML for network={network}, station={station}: {e}")

# Example usage
networks = ["TA", "BK"]
stations = ["A14A", "B15A"]

for network in networks:
    for station in stations:
        download_xml(network, station)
```
