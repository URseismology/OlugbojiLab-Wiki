```python
# Import necessary libraries
import obspy
from obspy import read, UTCDateTime
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Define time range for data retrieval
t1 = UTCDateTime("2023-01-01T00:00:00")
t2 = UTCDateTime("2023-01-02T00:00:00")

# Retrieve waveform data using the IRIS client
client = obspy.clients.fdsn.Client("IRIS")
try:
    st = client.get_waveforms(network="*", station="*", location="*", channel="*", starttime=t1, endtime=t2)
except Exception as e:
    print(f"Error retrieving data: {e}")
    st = obspy.Stream()

# Check the retrieved data
if not st.traces:
    print("No data retrieved. Stream is empty.")
else:
    # Convert to DataFrame for easier manipulation
    metadata = pd.DataFrame({
        'Network': [tr.stats.network for tr in st],
        'Station': [tr.stats.station for tr in st],
        'Channel': [tr.stats.channel for tr in st],
        'Start Time': [tr.stats.starttime for tr in st],
        'End Time': [tr.stats.endtime for tr in st]
    })

    # Display metadata
    print(metadata)

# Visualization (if data is retrieved)
if not st.traces:
    print("No waveforms to visualize.")
else:
    for tr in st:
        plt.figure(figsize=(10, 5))
        plt.plot(tr.times(), tr.data)
        plt.title(f"Waveform for {tr.stats.station} - {tr.stats.channel}")
        plt.xlabel("Time (s)")
        plt.ylabel("Amplitude")
        plt.show()
```
